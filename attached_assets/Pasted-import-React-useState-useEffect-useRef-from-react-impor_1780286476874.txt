import React, { useState, useEffect, useRef } from 'react';
import { base44 } from '@/api/base44Client';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { Plus, MessageSquare, Loader2, Trash2 } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { ScrollArea } from '@/components/ui/scroll-area';
import { cn } from '@/lib/utils';
import MessageBubble from '@/components/chat/MessageBubble';
import ChatInput from '@/components/chat/ChatInput';
import { motion, AnimatePresence } from 'framer-motion';

export default function Chat() {
  const [activeConversationId, setActiveConversationId] = useState(null);
  const [agentConversation, setAgentConversation] = useState(null);
  const [messages, setMessages] = useState([]);
  const [sending, setSending] = useState(false);
  const [showSidebar, setShowSidebar] = useState(false);
  const bottomRef = useRef(null);
  const queryClient = useQueryClient();

  const { data: conversations = [], isLoading: loadingConvos } = useQuery({
    queryKey: ['conversations'],
    queryFn: () => base44.entities.Conversation.list('-created_date', 50),
    initialData: [],
  });

  // Load agent conversation when selected
  useEffect(() => {
    if (!activeConversationId) {
      setMessages([]);
      setAgentConversation(null);
      return;
    }
    const convo = conversations.find(c => c.id === activeConversationId);
    if (convo?.agent_conversation_id) {
      base44.agents.getConversation(convo.agent_conversation_id).then(ac => {
        setAgentConversation(ac);
        setMessages(ac.messages || []);
      });
    }
  }, [activeConversationId, conversations]);

  // Subscribe to real-time updates
  useEffect(() => {
    if (!agentConversation?.id) return;
    const unsub = base44.agents.subscribeToConversation(agentConversation.id, (data) => {
      setMessages(data.messages || []);
    });
    return () => unsub();
  }, [agentConversation?.id]);

  // Auto scroll
  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleNewConversation = async () => {
    const ac = await base44.agents.createConversation({
      agent_name: 'madetvceo',
      metadata: { name: 'New Conversation' },
    });
    const record = await base44.entities.Conversation.create({
      title: 'New Conversation',
      agent_conversation_id: ac.id,
      message_count: 0,
    });
    queryClient.invalidateQueries({ queryKey: ['conversations'] });
    setActiveConversationId(record.id);
    setAgentConversation(ac);
    setMessages([]);
    setShowSidebar(false);
  };

  const handleSend = async (text, fileUrls = []) => {
    if (!agentConversation) return;
    setSending(true);
    const msgPayload = { role: 'user', content: text };
    if (fileUrls.length > 0) msgPayload.file_urls = fileUrls;
    await base44.agents.addMessage(agentConversation, msgPayload);

    // Update conversation record
    const convo = conversations.find(c => c.id === activeConversationId);
    if (convo) {
      await base44.entities.Conversation.update(convo.id, {
        last_message: text.substring(0, 100),
        message_count: (convo.message_count || 0) + 1,
        title: convo.title === 'New Conversation' && text.length > 3 ? text.substring(0, 40) : convo.title,
      });
      queryClient.invalidateQueries({ queryKey: ['conversations'] });
    }

    // Log the action
    await base44.entities.AgentLog.create({
      action_type: 'chat',
      summary: text.substring(0, 100),
      status: 'success',
    });
    queryClient.invalidateQueries({ queryKey: ['agent-logs'] });
    setSending(false);
  };

  const handleDelete = async (id) => {
    await base44.entities.Conversation.delete(id);
    if (activeConversationId === id) {
      setActiveConversationId(null);
    }
    queryClient.invalidateQueries({ queryKey: ['conversations'] });
  };

  return (
    <div className="flex h-[calc(100vh-57px)] lg:h-screen">
      {/* Conversation List */}
      <div className={cn(
        "fixed lg:static inset-0 z-40 w-72 bg-card border-r border-border flex flex-col transition-transform duration-300",
        showSidebar ? "translate-x-0" : "-translate-x-full lg:translate-x-0"
      )}>
        <div className="p-4 border-b border-border flex items-center justify-between">
          <h2 className="font-semibold text-sm">Conversations</h2>
          <Button size="sm" variant="ghost" className="text-primary" onClick={handleNewConversation}>
            <Plus className="w-4 h-4 mr-1" /> New
          </Button>
        </div>
        <ScrollArea className="flex-1">
          <div className="p-2 space-y-1">
            {conversations.map(c => (
              <div
                key={c.id}
                className={cn(
                  "flex items-center gap-2 px-3 py-2.5 rounded-lg cursor-pointer text-sm transition-colors group",
                  activeConversationId === c.id ? "bg-primary/10 text-primary" : "text-muted-foreground hover:bg-secondary"
                )}
                onClick={() => { setActiveConversationId(c.id); setShowSidebar(false); }}
              >
                <MessageSquare className="w-3.5 h-3.5 shrink-0" />
                <span className="flex-1 truncate">{c.title || 'Untitled'}</span>
                <button className="opacity-0 group-hover:opacity-100 transition-opacity" onClick={(e) => { e.stopPropagation(); handleDelete(c.id); }}>
                  <Trash2 className="w-3 h-3 text-muted-foreground hover:text-destructive" />
                </button>
              </div>
            ))}
          </div>
        </ScrollArea>
      </div>

      {/* Mobile sidebar toggle */}
      {showSidebar && <div className="fixed inset-0 bg-black/60 z-30 lg:hidden" onClick={() => setShowSidebar(false)} />}

      {/* Chat Area */}
      <div className="flex-1 flex flex-col">
        {!activeConversationId ? (
          <div className="flex-1 flex items-center justify-center p-8">
            <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="text-center max-w-md">
              <div className="w-16 h-16 rounded-2xl bg-primary/20 flex items-center justify-center mx-auto mb-6 glow-blue">
                <MessageSquare className="w-8 h-8 text-primary" />
              </div>
              <h2 className="text-2xl font-bold mb-2">MadeTVCEO Agent</h2>
              <p className="text-muted-foreground mb-6 text-sm">
                Ask me anything. Build apps, generate images, research topics, scan for threats — I'm ready.
              </p>
              <Button onClick={handleNewConversation} className="bg-primary hover:bg-primary/90">
                <Plus className="w-4 h-4 mr-2" /> Start Conversation
              </Button>
              <Button variant="ghost" className="ml-2 lg:hidden" onClick={() => setShowSidebar(true)}>
                View History
              </Button>
            </motion.div>
          </div>
        ) : (
          <>
            {/* Header */}
            <div className="px-4 py-3 border-b border-border flex items-center gap-3 bg-card">
              <Button variant="ghost" size="sm" className="lg:hidden" onClick={() => setShowSidebar(true)}>
                <MessageSquare className="w-4 h-4" />
              </Button>
              <div className="w-2 h-2 rounded-full bg-accent" />
              <span className="text-sm font-medium truncate">
                {conversations.find(c => c.id === activeConversationId)?.title || 'Chat'}
              </span>
            </div>

            {/* Messages */}
            <ScrollArea className="flex-1 p-4 lg:p-6">
              <div className="max-w-3xl mx-auto space-y-4">
                <AnimatePresence>
                  {messages.map((msg, i) => (
                    <motion.div key={i} initial={{ opacity: 0, y: 5 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: i * 0.02 }}>
                      <MessageBubble message={msg} />
                    </motion.div>
                  ))}
                </AnimatePresence>
                {sending && (
                  <div className="flex items-center gap-2 text-muted-foreground text-sm">
                    <Loader2 className="w-4 h-4 animate-spin text-primary" />
                    <span>Processing...</span>
                  </div>
                )}
                <div ref={bottomRef} />
              </div>
            </ScrollArea>

            {/* Input */}
            <div className="max-w-3xl mx-auto w-full">
              <ChatInput onSend={handleSend} disabled={sending} />
            </div>
          </>
        )}
      </div>
    </div>
  );
}
